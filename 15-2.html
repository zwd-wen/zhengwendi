JavaScript 程序结构：条件与循环

郑雯迪 · 2026-10-26

一、条件语句

if / else

var yl_age = 20;
if (yl_age >= 18) {
    "成年人";
} else {
    "未成年";
}

if / else if / else

var yl_score = 85;
if (yl_score >= 90) "A";
else if (yl_score >= 80) "B";
else if (yl_score >= 60) "C";
else "D";

switch

var yl_day = 3;
switch (yl_day) {
  case 1: "周一"; break;
  case 3: "周三"; break;
  default: "其他";
}

三元运算符

var yl_num = 7;
(yl_num % 2 === 0)
  ? "偶数" : "奇数";

二、循环语句

for 循环

var yl_sum = 0;
for (var i = 1; i <= 5; i++) {
    yl_sum += i;
}
// 1+2+3+4+5

while 循环

var yl_n = 1, yl_fact = 1;
while (yl_n <= 5) {
    yl_fact *= yl_n;
    yl_n++;
}
// 5 的阶乘

do...while 循环

var yl_x = 0, yl_out = "";
do {
    yl_out += yl_x + " ";
    yl_x++;
} while (yl_x < 4);

for...in 遍历对象

var yl_obj = {
  姓名: "郑雯迪",
  年龄: 20
};
for (var k in yl_obj) {
  k + ": " + yl_obj[k];
}

for...of 遍历数组

var yl_arr = ["HTML","CSS","JS"];
for (var v of yl_arr) {
    v;
}

break 与 continue

for (var i = 1; i <= 10; i++) {
  if (i === 5) break;   // 遇到5跳出
  if (i % 2 === 0) continue; // 偶数跳过
  // 输出 1, 3
}
