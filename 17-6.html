核心代码（Promise + async/await）

// 用 setTimeout 模拟每个动作的耗时，返回一个 Promise
function delay(ms) {
    return new Promise(function (resolve) {
        setTimeout(resolve, ms);
    });
}

async function makeDumplings() {
    // ① 线上、线下同时买，谁先回来用谁（race：只取最快的结果）
    var winner = await Promise.race([
        delay(4000).then(function () { return '线上购买'; }),
        delay(6000).then(function () { return '线下购买'; })
    ]);
    console.log(winner + ' 先完成，原料到手');

    // ② 取原料
    await delay(2000);

    // ③ 和面、和馅同时进行，两件都完成才继续（all：等待全部完成）
    await Promise.all([ delay(3000), delay(4000) ]);

    // ④ 包饺子 → ⑤ 煮饺子 → ⑥ 吃饺子，依次执行
    await delay(5000);
    await delay(4000);
    await delay(2000);
    console.log('开吃！');
}
